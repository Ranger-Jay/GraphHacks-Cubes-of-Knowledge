// TEST-ONLY harness (Cora, NOT Mara's adapter): PR #50 contract routes over the REAL graph module (366eb56) on a disposable
// FalkorDBLite. No tokens; owner role is assumed unless /__ctl sets deny. Fault injection via POST /__ctl (JSON).
import http from 'node:http';
import { readFile } from 'node:fs/promises';
import { join } from 'node:path';
import { createHash } from 'node:crypto';
import { GraphClient, configFromEnv } from '../packages/graph/src/index.ts';
import { importIdea, importStatus, ideaverseOverview, loadIdeaManifest, seedIdeaverse } from '../packages/graph/src/ideaverse.ts';

const root = process.cwd();
const tenant = 't-wad';
const client = new GraphClient(configFromEnv());
const ctl = { deny: false, delayImportMs: 0, dropReplyAfterCommit: 0, bumpBeforeImport: '' as string, denyImport: false, lieOverview: 0 };
const log: Array<{ n: number; method: string; path: string; cache: string | undefined; pragma: string | undefined; hasAuth: boolean; browser: boolean; body?: unknown; status?: number }> = [];
let n = 0;
const send = (res: http.ServerResponse, status: number, body: unknown) => { res.writeHead(status, { 'Content-Type': 'application/json', 'Cache-Control': 'no-store' }); res.end(JSON.stringify(body)); };
const readBody = (req: http.IncomingMessage) => new Promise<string>(r => { let b = ''; req.on('data', c => { b += c; if (b.length > 16384) req.destroy(); }); req.on('end', () => r(b)); });
const KEY = /^[A-Za-z0-9_-]{8,64}$/;

async function reconcile(ideaId: string, receipt: { sha256: string; membershipId: string; importedAt: string; workspaceRevision: number; workspaceId: string; ideaId: string }) {
  const ov = await ideaverseOverview(client, root, tenant);
  if (ov.status !== 'AVAILABLE') return false;
  const listed = ov.ideas.filter(i => i.id === ideaId);
  return listed.length === 1 && listed[0]!.imported && listed[0]!.membershipId === receipt.membershipId && listed[0]!.importedAt === receipt.importedAt && listed[0]!.sha256 === receipt.sha256
    && !ov.integrity.some(f => f.id === ideaId) && ov.workspace.revision >= receipt.workspaceRevision && ov.workspace.id === receipt.workspaceId;
}

http.createServer(async (req, res) => {
  const url = new URL(req.url ?? '/', 'http://x');
  const rec = { n: ++n, method: req.method ?? '', path: url.pathname, cache: req.headers['cache-control'], pragma: req.headers['pragma'], hasAuth: 'authorization' in req.headers, browser: !!req.headers['referer'] } as (typeof log)[number];
  if (!url.pathname.startsWith('/__ctl')) log.push(rec);
  try {
    if (url.pathname === '/__ctl' && req.method === 'POST') { Object.assign(ctl, JSON.parse(await readBody(req))); return send(res, 200, ctl); }
    if (url.pathname === '/__ctl/log') return send(res, 200, log);
    if (url.pathname === '/__ctl/reset-log') { log.length = 0; return send(res, 200, {}); }
    if (url.pathname === '/__ctl/seed') { const r = await seedIdeaverse(client, root, { tenantId: tenant }); return send(res, 200, r); }
    if (url.search) return send(res, 400, { status: 'REJECTED' });
    const m = url.pathname.match(/^\/api\/studio\/ideas(?:\/([a-z0-9][a-z0-9-]{0,63}))?(\/import|\/import-status)?$/);
    if (!m) return send(res, 404, { error: 'not found' });
    const [, ideaId, suffix] = m;
    if (ctl.deny) { rec.status = 403; return send(res, 403, { error: { code: 'FORBIDDEN' } }); }
    if (!ideaId && req.method === 'GET') {
      const ov = await ideaverseOverview(client, root, tenant);
      rec.status = ov.status === 'UNAVAILABLE' ? 503 : 200;
      return send(res, rec.status, ov);
    }
    if (ideaId && !suffix && req.method === 'GET') {
      const ov = await ideaverseOverview(client, root, tenant);
      if (ov.status === 'UNAVAILABLE') return send(res, 503, { status: 'UNAVAILABLE', reason: ov.reason });
      if (ov.status === 'EMPTY' || !ov.ideas.some(i => i.id === ideaId) || ov.integrity.some(f => f.id === ideaId)) return send(res, 404, { status: 'NOT_FOUND' });
      const manifest = await loadIdeaManifest(root); const e = manifest.ideas.find(i => i.id === ideaId)!;
      const bytes = await readFile(join(root, e.path));
      if (createHash('sha256').update(bytes).digest('hex') !== e.sha256) return send(res, 503, { status: 'UNAVAILABLE', reason: 'SOURCE_MISMATCH' });
      return send(res, 200, { status: 'AVAILABLE', idea: { ...ov.ideas.find(i => i.id === ideaId)!, text: bytes.toString('utf8').trim() } });
    }
    if (ideaId && suffix && req.method === 'POST') {
      let body: Record<string, unknown>;
      try { body = JSON.parse(await readBody(req)); } catch { return send(res, 400, { status: 'REJECTED', reason: 'INVALID_REQUEST' }); }
      const keys = Object.keys(body).sort().join(',');
      if (keys !== 'expectedRevision,idempotencyKey' || typeof body.idempotencyKey !== 'string' || !KEY.test(body.idempotencyKey) || !Number.isSafeInteger(body.expectedRevision) || (body.expectedRevision as number) < 0) return send(res, 400, { status: 'REJECTED', reason: 'INVALID_REQUEST' });
      rec.body = { ideaId, key: body.idempotencyKey, expectedRevision: body.expectedRevision };
      const key = body.idempotencyKey as string; const expectedRevision = body.expectedRevision as number;
      if (suffix === '/import-status') {
        const s = await importStatus(client, tenant, key);
        if (s.status === 'UNAVAILABLE') return send(res, 503, s);
        if (s.status !== 'IMPORTED') { rec.status = 200; return send(res, 200, { status: 'NOT_FOUND' }); }
        if (s.receipt.ideaId !== ideaId) return send(res, 409, { status: 'CONFLICT', reason: 'KEY_REUSED' });
        if (!(await reconcile(ideaId, s.receipt))) return send(res, 503, { status: 'UNCONFIRMED', reason: 'READBACK_MISMATCH' });
        return send(res, 200, { status: 'IMPORTED', replayed: true, receipt: s.receipt });
      }
      if (ctl.denyImport && suffix === '/import') { rec.status = 403; return send(res, 403, { error: { code: 'FORBIDDEN' } }); }
      if (ctl.delayImportMs) await new Promise(r => setTimeout(r, ctl.delayImportMs));
      if (ctl.bumpBeforeImport) { const other = ctl.bumpBeforeImport; ctl.bumpBeforeImport = ''; const ov = await ideaverseOverview(client, root, tenant); if (ov.status === 'AVAILABLE') await importIdea(client, root, { tenantId: tenant, ideaId: other, idempotencyKey: `external-${other}-bump`, expectedRevision: ov.workspace.revision }); }
      const r = await importIdea(client, root, { tenantId: tenant, ideaId, idempotencyKey: key, expectedRevision });
      if (ctl.dropReplyAfterCommit > 0 && r.status === 'IMPORTED') { ctl.dropReplyAfterCommit--; rec.status = 0; req.socket.destroy(); return; }
      if (r.status === 'IMPORTED') {
        const ok = await reconcile(ideaId, r.receipt);
        rec.status = ok ? 200 : 503;
        return ok ? send(res, 200, r) : send(res, 503, { status: 'UNCONFIRMED', reason: 'READBACK_MISMATCH' });
      }
      if (r.status === 'CONFLICT') { rec.status = 409; return send(res, 409, r); }
      if (r.status === 'REJECTED') { rec.status = 422; return send(res, 422, r); }
      rec.status = 503; return send(res, 503, r);
    }
    send(res, 405, { error: 'method' });
  } catch (e) { rec.status = 500; send(res, 503, { status: 'UNAVAILABLE', reason: 'QUERY_FAILED' }); }
}).listen(8788, '127.0.0.1', () => console.log('studio harness on 8788'));
