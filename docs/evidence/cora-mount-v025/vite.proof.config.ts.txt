import { defineConfig, loadEnv } from 'vite';

// SCRATCH proof config: identical to vite.config.ts plus /api/studio -> the test-only studio harness (8788). Not committed.
export default defineConfig(({ mode }) => {
  const env = loadEnv(mode, repoRoot, 'API_');

    '/api/studio': { target: 'http://127.0.0.1:8788', changeOrigin: false },
    '/api': { target: 'http://127.0.0.1:8787', changeOrigin: false, configure: bearerInjector(readerTokenFrom(env)) },
  } } };
});
